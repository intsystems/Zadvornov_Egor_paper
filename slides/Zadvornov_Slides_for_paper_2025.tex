\begin{frame}{Прогнозирование высоковолатильных временных рядов}
  \vspace{-0.4cm}
    \begin{block}{Цель исследования}
          \vspace{-0.1cm}
        Повысить точность прогноза распределений тематик в текстовых документах, распеределённых во времени, с помощью метода, устойчивого к циклическим\footnote{Под циклом подразумевается изменение с переменным периодом, непривязанное ко времени, которое не портит предположение о слабой стационарности временного ряда. Отражают компоненту цикличности с переменным периодом, увеличивают дисперсию ошибки прогноза.} словам. 
    \end{block}
  \vspace{-0.2cm}
    \begin{block}{Проблема}
      \vspace{-0.1cm}
        Классические регрессионные (AR) подходы прогнозирования распределения тематик неустойчивы к появлению нерегулярных циклических слов тематик.
    \end{block}
  \vspace{-0.2cm}
    \begin{block}{Предлагаемое решение}
          \vspace{-0.1cm}
        Предлагается повысить точноть AR-моделей с помощью выделения подмножества слов словаря, коррелирующих с временным распределением соответствующих тематик.
    \end{block}
\end{frame}



\begin{frame}{Обозначения и исходные данные}
  \vspace{-0.3cm}
  \begin{block}{Обозначения и исходные данные}
    % \vspace{-0.4cm}
    Рассматривается конечное множество тематик (кластеров) $\overline{T} = \{T_1, \dots, T_K\}$ и словарь $\mathcal{W}_{\mathrm{all}}$, содержащий $W = |\mathcal{W}_{\mathrm{all}}|$ слов. Каждой тематике $T_k$ соответствует подмножество слов $T_k \subset \mathcal{W}_{\mathrm{all}}$. 

    Для каждого слова $w$ и момента времени $t$ задана частота:
    \footnotesize
    \(
    f(w,t)\colon \mathcal{W}_{\mathrm{all}} \times \{0, \dots, t{-}1\} \to \mathbb{N}_{\ge 0}
    \)

    $K$ — число тематик, $T_k$~— тематика, $f(w,t)$~— частота слова $w$ в момент $t$.
        \normalsize
  \end{block}

  % \vspace{-0.3cm}
  \begin{block}{Целевая переменная - распределение тематик во времени}
    \vspace{-0.4cm}
    \footnotesize
    \[
    \nu_k(t) := \frac{\sum_{w \in T_k} f(w,t)}{\sum_{j=1}^K \sum_{w \in T_j} f(w,t)},\quad \overline{\nu}(t) := (\nu_1(t), \dots, \nu_K(t)),\quad \sum_{k=1}^K \nu_k(t) = 1
    \]
    \normalsize
  \end{block}
\end{frame}

  \vspace{-0.3cm}
  \begin{block}{Задача}
    % \vspace{-0.5cm}
    Минимизация квадратичной ошибки прогноза распределения тематик: 
       \footnotesize
    \[
    \min_{\overline{\phi} \in \mathbb{R}^d} \; 
    \sum_{t} 
    \left\|
    \overline{\nu}(t)
    -

    \right\|_2^2
    \]
        \normalsize
    где $\widehat{\overline{\nu}}$ --- прогноз распределения тематик на момент $t$ на основе предыстории длины $t_0$.

  \end{block}

  % \vspace{-0.2cm}
  \begin{block}{Цель}
    Построить устойчивую модель прогноза распределения частот тематик в условиях аномальных значений и цикличных изменений.
  \end{block}

\begin{frame}[t]{Построение тематических кластеров из словаря}
\small
\vspace{-0.3cm}
\begin{block}{Постановка задачи кластеризации}
    % \vspace{-0.4cm}
\footnotesize
\(\mathcal{W}_{\mathrm{all}}=\{w_1,\dots,w_W\}\).  
Найти решение \(\overline{T}\), \(T_i\cap T_j=\varnothing\),
\(\bigsqcup_kT_k=\mathcal{W}_{\mathrm{all}}\).
\normalsize
\end{block}

% \vspace{-0.3cm}
\begin{block}{Существущие методы решения}
\vspace{-0.2cm}
\begin{itemize}
    \footnotesize
  \item \emph{Embeddings+K-Means:}
    \(\displaystyle\min_{T_k}\sum_{k=1}^K\sum_{w\in T_k}\|\mathbf e_w-\mu_k\|^2,\;\mathbf e_w=\mathrm E(w).\)
  \item \emph{Latent Dirichlet Allocation(LDA):} 
    $\displaystyle p(\theta,\phi,z,w|\alpha,\beta)
            =\prod_{k=1}^K p(\phi_k|\beta)\prod_{d=1}^D p(\theta_d|\alpha)
             \prod_{n=1}^{N_d}p(z_{dn}|\theta_d)p(w_{dn}|\phi_{z_{dn}}).$
    \normalsize        
\end{itemize}
\end{block}

\vspace{-0.7cm}
\begin{block}{Оптимизация \(K\) - числа тематик.}
% \vspace{-0.4cm}
\footnotesize
\(\displaystyle K^*=\arg\min_K\frac1K\sum_{m=1}^KC_{\mathrm{cv}}(c_m),\)  
где \(C_{\mathrm{cv}}\) — мера когерентности тем.

\(
  C_{\mathrm{cv}}
  \;=\;
  \frac{1}{\lvert S^{\mathrm{one}}_{\mathrm{set}}\rvert}
  \sum_{(W_0,W_*)\in S^{\mathrm{one}}_{\mathrm{set}}}
    \tilde m_{\cos(\mathrm{nlr},1)}(W_0,W_*)
\)
\vspace{0.2cm}
где 

\(
  S^{\mathrm{one}}_{\mathrm{set}}
  =\{\,(\{w_i\},W)\mid w_i\in W\subset\mathcal W_{\mathrm{all}}\}
\) и 
\(\tilde m_{\cos(\mathrm{nlr},1)}\) — косинусное сходство между векторами множеств \(W_0\) и \(W_*\).
\normalsize
\end{block}

\end{frame}

\begin{frame}{Классический подход к прогнозу тематик}
\begin{block}{Авторегрессионная модель}
Для каждой тематики $T_k\in\overline{T}$ используется модель вида:
\footnotesize
\[
\widehat{\nu}_k(t+1) = g_{\psi_k}(\nu_k(t), \dots, \nu_k(t - t_0 + 1))
\]
\normalsize
где $g_{\psi_k}$ — модель предсказания тематик, прогнозирующая динамику каждой темы независимо.
\end{block}

\begin{block}{Прогноз распределения тематики}
Условное математическое ожидание следующего значения распределения при заданной предыстории:
\footnotesize
\[
\widehat{\nu}_k(t+1)
\;:=\;
\mathbb{E}_{\widehat{\psi}_k}\!\bigl[\nu_k(t+1)\mid\mathcal F_t\bigr].
\]
\normalsize
\end{block}

\vspace{-0.5cm}
\begin{block}{Функционал ошибки}
Среднеквадратичное отклонение прогнозов от фактических значений распределения тем:
\footnotesize
\[
\mathcal{L} = \sum_{t}
\|\overline{\nu}(t) - \widehat{\overline{\nu}}(t)\|_2^2
\]
\normalsize
\end{block}
\end{frame}

\begin{frame}{Ограничения подхода в условиях высокой волатильности}
\begin{block}{Выделяемые подмножества слов исходного словаря и связь с теорией временных рядов}
% \vspace{-0.3cm}
\begin{enumerate}
\footnotesize
  \item $\mathcal{W}_{\mathrm{core}}$ --- слова с устойчивой временной структурой и высокой автокорреляцией: тренды (плавные долгосрочные изменения) и сезонные компоненты (периодические колебания с постоянным периодом):
\begin{flalign*}
& \mathcal{W}_{\mathrm{core}} = \left\{ w \in \mathcal{W}_{\mathrm{all}} \;\middle|\; \rho_w := \operatorname{Corr}(f(w,t),f(T_k,t+1))\ge r, \text{ где } w \in T_k \right\} &
\end{flalign*}

\item $\mathcal{W}_{\mathrm{WN}}$ --- мультитематические слова: моделируются случайной ошибкой из-за высокой вероятности перехода между тематиками:
\begin{flalign*}
& \mathcal{W}_{\mathrm{WN}} = \left\{ w \in \mathcal{W}_{\mathrm{all}} \;\middle|\; f(w,t) = [\varepsilon_w(t)]_+,\, \rho_w = 0 \right\} &
\end{flalign*}

\item $\mathcal{W}_{\mathrm{SD}}$ --- циклы: изменения с переменным периодом. Сохраняют стационарность, но увеличивает дисперсию ошибки прогноза:
\begin{flalign*}
& \mathcal{W}_{\mathrm{SD}} = \left\{ w \in \mathcal{W}_{\mathrm{all}} \;\middle|\; f(w,t) = [S_w(t) + \varepsilon_w(t)]_+, \quad \rho_w(\tau) \approx 0\right\}, & \\
& S_w(t) = \textstyle\sum A_{w,m} \kappa(t{-}\tau_{w,m}),  \kappa(\tau)=\frac{\mathrm e^{-\lambda|\tau|}}{1+\tau^{2}} &
\end{flalign*}
\end{enumerate}
\normalsize
\end{block}
\end{frame}

\begin{frame}{Влияние шумовых слов на прогноз тематик}
\begin{block}{Утверждение 1 (Задворнов, 2025)}
\footnotesize
Пусть \(
\mathcal{W}_{\mathrm{all}} = \mathcal{W}_{\mathrm{core}} \cup \mathcal{W}_{\mathrm{WN}} \cup \mathcal{W}_{\mathrm{SD}}, \quad \mathcal{W}_a \cap \mathcal{W}_b = \varnothing
\)
при условии попарной некоррелированности компонент
\[
\forall\,a \neq b,\;\forall\,\tau \in \mathbb{Z}:\quad
Cov\bigl(f^{(a)}(T_k,t),\,f^{(b)}(T_k,t+\tau)\bigr) = 0
\]
Тогда
\[
Cov\bigl(f(T_k,t),\,f(T_k,t+1)\bigr) = Cov\bigl(f^{(\mathrm{core})}(t),\,f^{(\mathrm{core})}(t{+}1)\bigr)
\]
и
\[
\rho_k(1) = \frac{Cov^{\mathrm{core}}}{Var^{\mathrm{core}} + \sigma_{\mathrm{WN}}^2 + \sigma_{\mathrm{SD}}^2}.
\]
\normalsize
\end{block}

\vspace{-0.4cm}
\begin{block}{Вывод}
\begin{enumerate}
\item В условиях высокой волатильности $\mathcal{W}_{\mathrm{WN}} \cup \mathcal{W}_{\mathrm{SD}}$ понижают автокорреляцию $\rho_k(1)$, нарушая предпосылки AR-моделей о существенной корреляционной зависимости, и уменьшают точность прогноза. 
\item Необходим подход, устойчивый к $\mathcal{W}_{\mathrm{WN}}\cup\mathcal{W}_{\mathrm{SD}}$.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Предлагается алгоритм ARFilter}

\begin{block}{Принцип работы}
% \footnotesize
\begin{enumerate}
\item Формируется вектор корреляций между частотами слов и будущими значениями соответствующих тематик.  

\item Выделяется набор слов $\mathcal{W}_{\mathrm{core}}$ с устойчивой временной структурой.  

\item Для каждого выделенного слова строится отдельная AR-модель, прогнозирующая относительные доли:
\footnotesize
\[s(w,t) = \frac{f(w,t)}{\sum_{v\in\mathcal{W}_{\mathrm{core}}}f(v,t)}\]
\normalsize

\item Прогноз относительной доли тематики вычисляется как сумма прогнозов слов, входящих в неё.  

% \item Весовые коэффициенты слов адаптивно корректируются на основе дисперсий их прогнозов.  

% \normalsize
\end{enumerate}
\end{block}
\end{frame}


\begin{frame}{Теоретическое обоснование эффективности}
\begin{block}{Теорема (Задворнов, 2025)}
\footnotesize
Пусть для тематики $T_k$:
\begin{enumerate}
    \item $\mu_k = \mathbb{E}[\nu_k(t) \mid \mathcal{F}_{t-1}]$ - условное матожидание распределения по всем словам
    \item $\mu_k^{(\mathrm{core})} = \mathbb{E}[\widehat{\nu}_k^{(\mathrm{core})}(t) \mid \mathcal{F}_{t-1}]$ - условное матожидание по core-словам
    \item Выполнены условия Утверждения 1 о разбиении словаря
\end{enumerate}
и дополнительно выполнены условия на шумовые слова:
\begin{enumerate}
    \item Для $w \in \mathcal{W}_{\mathrm{WN}}$: $\mathbb{E}[f(w,t) \mid \mathcal{F}_{t-1}] \leq C_{\mathrm{WN}} \sigma_{\mathrm{WN}}$ 
    \item Для $w \in \mathcal{W}_{\mathrm{SD}}$: $\mathbb{E}[f(w,t) \mid \mathcal{F}_{t-1}] \leq C_{\mathrm{SD}} \sigma_{\mathrm{SD}}$ (в отсутствие пиков)
    \item $|\mathcal{W}_{\mathrm{WN}}| C_{\mathrm{WN}}^2 + |\mathcal{W}_{\mathrm{SD}}| C_{\mathrm{SD}}^2 < \kappa$
\end{enumerate}
где $\kappa > 0$ - минимальное собственное значение информационной матрицы модели прогноза.

Тогда:
\[
  \mathbb{E}\!\bigl[\mathcal L_{\mathrm{ARFilter}}\mid \mathcal{F}_{t}\bigr]:=\mathrm{MSE}\bigl(\widehat{\nu}_k^{(\mathrm{core})}(t)\bigr)
  \;<\; 
  \mathrm{MSE}\bigl(\widehat{\nu}_k^{(\mathrm{all})}(t)\bigr) =:\mathbb{E}\!\bigl[\mathcal L_{\mathrm{baseline}}\mid \mathcal{F}_{t}\bigr]
\]
\normalsize
\end{block}

\end{frame}



\begin{frame}{Псевдокод алгоритма ARFilter}
\vspace{-0.4cm}
\begin{figure}[h]
  \centering
  \begin{minipage}{.9\linewidth}
    \begin{algorithm}[H]
      \caption{ARFilter}\label{alg:selection}
      \begin{algorithmic}[1]
        \scriptsize
        \Require Корпус \(\mathcal{D}=\{(d_i,t_i)\}_{i=1}^N\), порог \(r\), окно \(t_0\)
        \Ensure \(\overline{T},\,\overline{\theta},\,\overline{\psi}\)
        \State Извлечь \(\mathcal{W}_{\mathrm{all}}\) и частоты \(f(w,t)\) по $\mathcal{D}$
        \State \(\overline{T}=\{T_k\}\) с помощью вероятностной модели
        \ForAll{\(w\in\mathcal{W}_{\mathrm{all}}\)}
          \State \(\widehat\rho_w\gets\mathrm{Corr}\bigl(f(w,t),f(T_k,t+1)\bigr),\text{ где } w \in T_k\)
          \State \(\Theta_w\gets\mathbb I[\widehat\rho_w\ge r]\)
        \EndFor
        \State \(\mathcal{W}_{\mathrm{core}}\gets\{w:\Theta_w=1\}\)
        \ForAll{\(w\in\mathcal{W}_{\mathrm{core}}\)}
          \State \(s(w,t)\gets \frac{f(w,t)}{\sum_{v\in\mathcal{W}_{\mathrm{core}}}f(v,t)}\)
          \State \(\psi_w\gets\arg\min_{\psi}\sum_{t\in T_{\mathrm{train}}}\bigl(s(w,t)-g_{\psi}(s(w,t-1:t-t_0))\bigr)^2\)
        \EndFor
        \State \Return \(\overline{T},\,\overline{\theta},\,\overline{\psi}\)
        \normalsize
      \end{algorithmic}
    \end{algorithm}
  \end{minipage}
\end{figure}

\end{frame}

\emph{Следствие:} Шумовые слова увеличивают дисперсию прогноза.

\emph{Следствие:} Алгоритм ARFilter превосходит классический подход.

\item[\textbf{D.}] Корреляционные коэффициенты \(\lvert\beta_w\rvert\): core-слова доминируют (зеленый), шумовые близки к нулю (серый). \\
\emph{Следствие:} Корреляционный алгоритм эффективно выделяет информативные признаки.
\end{enumerate}

\begin{frame}{Выносится на защиту}
\begin{enumerate}
    \item \textbf{Метод прогнозирования тематик во времени с использованием регуляризации при условии высокой волатильности}\\
    \begin{enumerate}
        \item Регуляризация через отбор признаков: \(\theta_w=\mathbb{I}[\operatorname{Corr}(f(w,t),f(T_k,t+1))\ge r], \text{ где } w \in T_k\)
        \item Локальные модели прогноза для отобранных признаков
    \end{enumerate}

    \item \textbf{Эмпирическое подтверждение эффективности}\\
    На синтетических данных достигнуто снижение MSE на 91\%:
    \[
        \Delta\mathrm{MSE} = \frac{\mathcal L_{\mathrm{baseline}}-\mathcal L_{\mathrm{ARFilter}}}{\mathcal L_{\mathrm{baseline}}}\times100\%
    \]

    \item \textbf{Условия из критерия применимости метода}\\
    Метод эффективен при условиях:
    \[
        \frac{\sigma_{\mathrm{noise}}^{2}}{\sigma_{\mathrm{core}}^{2}} \gg 1, \quad
        \rho_{w}(1)\approx0 \;\;\text{для}\;\; w\in\mathcal{W}_{\mathrm{WN}}\cup\mathcal{W}_{\mathrm{SD}}
    \]
\end{enumerate}
